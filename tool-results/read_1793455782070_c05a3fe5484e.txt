   464→   464→        const decoder = new TextDecoder()
   465→   465→        let buffer = ''
   466→   466→        let errorMsg: string | null = null
   467→   467→        let doneIds: {
   468→   468→          userMessageId: string
   469→   469→          assistantMessageId: string
   470→   470→          sessionId: string
   471→   471→          sessionTitle: string
   472→   472→        } | null = null
   473→   473→
   474→   474→        while (true) {
   475→   475→          const { done, value } = await reader.read()
   476→   476→          if (done) break
   477→   477→          buffer += decoder.decode(value, { stream: true })
   478→   478→          const parts = buffer.split('\n\n')
   479→   479→          buffer = parts.pop() ?? ''
   480→   480→          for (const part of parts) {
   481→   481→            const line = part.trim()
   482→   482→            if (!line.startsWith('data:')) continue
   483→   483→            const frame = JSON.parse(line.slice(5).trim())
   484→   484→            if (frame.t === 'delta') {
   485→   485→              pending += frame.v
   486→   486→              if (!flushTimer) flushTimer = setTimeout(flush, 60)
   487→   487→            } else if (frame.t === 'done') {
   488→   488→              doneIds = frame
   489→   489→            } else if (frame.t === 'error') {
   490→   490→              errorMsg = frame.v
   491→   491→            }
   492→   492→          }
   493→   493→        }
   494→   494→        if (flushTimer) {
   495→   495→          clearTimeout(flushTimer)
   496→   496→          flushTimer = null
   497→   497→        }
   498→   498→        flush()
   499→   499→
   500→   500→        if (errorMsg) throw new Error(errorMsg)
   501→   501→        if (doneIds) {
   502→   502→          const ids = doneIds
   503→   503→          setMessages((prev) =>
   504→   504→            prev.map((m) =>
   505→   505→              m.id === streamId
   506→   506→                ? { ...m, id: ids.assistantMessageId }
   507→   507→                : m.id === tempUserId
   508→   508→                  ? { ...m, id: ids.userMessageId }
   509→   509→                  : m
   510→   510→            )
   511→   511→          )
   512→   512→          // 会话可能被服务端自动命名/更新，刷新列表
   513→   513→          void refreshSessions()
   514→   514→        }
   515→   515→      } catch (error) {
   516→   516→        if (flushTimer) {
   517→   517→          clearTimeout(flushTimer)
   518→   518→          flushTimer = null
   519→   519→        }
   520→   520→        if (controller.signal.aborted) {
   521→   521→          // 用户点击「停止生成」：保留用户消息与已流出的部分内容
   522→   522→          flush()
   523→   523→          setMessages((prev) =>
   524→   524→            prev.filter((m) => m.id !== streamId || m.content.trim() !== '')
   525→   525→          )
   526→   526→          void refreshSessions()
   527→   527→          toast({ title: '已停止生成', description: '已生成的部分内容将保留' })
   528→   528→        } else {
   529→   529→          // 真实错误：丢弃占位消息，提示错误
   530→   530→          setMessages((prev) =>
   531→   531→            prev.filter((m) => m.id !== streamId && m.id !== tempUserId)
   532→   532→          )
   533→   533→          toast({
   534→   534→            title: 'AI 调用失败',
   535→   535→            description: error instanceof Error ? error.message : '请稍后重试',
   536→   536→            variant: 'destructive',
   537→   537→          })
   538→   538→        }
   539→   539→      } finally {
   540→   540→        abortRef.current = null
   541→   541→        setChatLoading(false)
   542→   542→        setStreamingId(null)
   543→   543→      }
   544→   544→    },
   545→   545→    [refreshSessions, toast]
   546→   546→  )
   547→   547→
   548→   548→  /** 停止当前 AI 生成（已生成的部分内容保留） */
   549→   549→  const stopGeneration = useCallback(() => {
   550→   550→    abortRef.current?.abort()
   551→   551→  }, [])

   556→   556→    try {

   558→   558→        method: 'DELETE',
   559→   559→      })
   560→   560→      const data = await res.json()
   561→   561→      if (!res.ok) throw new Error(data.error ?? '清空失败')
   562→   562→      setMessages(data.messages ?? [])
   563→   563→      void refreshSessions()
   564→   564→    } catch (error) {
   565→   565→      toast({
   566→   566→        title: '清空失败',
   567→   567→        description: error instanceof Error ? error.message : '请稍后重试',
   568→   568→        variant: 'destructive',
   569→   569→      })
   570→   570→    }
   571→   571→  }, [refreshSessions, toast])

   582→   582→      scheduleSave(tab.id, code)
   583→   583→      toast({
   584→   584→        title: '已应用到编辑器',
   585→   585→        description: '代码已写入当前文件，即将自动保存',
   586→   586→      })
   587→   587→    },
   588→   588→    [scheduleSave, toast]
   589→   589→  )
   590→   590→
   591→   591→  /** 从全局搜索跳转：打开文件并定位到指定行 */

   597→   597→    [openFile]
   598→   598→  )
   599→   599→
   600→   600→  // ===== 视图 =====

   621→   621→      jumpTarget={jumpTarget}
   622→   622→      onSave={() => {

   624→   624→        if (!tab) return
   625→   625→        if (saveTimersRef.current[tab.id]) {

   627→   627→          delete saveTimersRef.current[tab.id]
   628→   628→        }
   629→   629→        void saveTab(tab.id)
   630→   630→      }}
   631→   631→    />
   632→   632→  )
   633→   633→

   643→   643→      onCreateSession={createSession}

   646→   646→      onSend={sendMessage}
   647→   647→      onStop={stopGeneration}
   648→   648→      onClear={clearChat}
   649→   649→      onApplyCode={applyCode}

   651→   651→  )
   652→   652→
   653→   653→  /** 按目录分组供快速打开面板使用 */
   654→   654→  const quickOpenView = (

   656→   656→      <CommandInput placeholder="输入文件名搜索…" />
   657→   657→      <CommandList>
   658→   658→        <CommandEmpty>没有匹配的文件</CommandEmpty>
   659→   659→        <CommandGroup heading="打开文件">
   660→   660→          {files.map((f) => (
   661→   661→            <CommandItem
   662→   662→              key={f.id}
   663→   663→              value={f.path}
   664→   664→              onSelect={() => {
   665→   665→                setQuickOpen(false)
   666→   666→                void openFile(f.id)
   667→   667→              }}
   668→   668→            >
   669→   669→              <FileCode2 className="h-4 w-4 text-clay" />
   670→   670→              <span className="truncate font-mono text-[13px]">{f.path}</span>
   671→   671→            </CommandItem>

   676→   676→  )
   677→   677→
   678→   678→  return (

   687→   687→      />
   688→   688→      {/* 顶栏 */}
   689→   689→      <header className="flex h-12 shrink-0 items-center gap-2.5 border-b border-border bg-card px-4">

   702→   702→          href="/api/export"
   703→   703→          download
   704→   704→          className="ml-auto flex items-center gap-1.5 rounded-md px-2 py-1 text-xs text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
